```js
/**
 * Pepper — CCH Studio's staff-only AI assistant. Drafts client copy and answers
 * "where do things stand" questions, grounded in this project's own activity
 * data and CCH's voice rules.
 *
 * NEW FILE. Does not modify cchFixItBot.js, aiInvoiceSummary.js, aiProgressUpdate.js,
 * or cchVoiceProfile.js. Follows the same pattern as cchFixItBot.js on purpose.
 * Staff-only, scope confirmed Jul 26: never called from the client portal.
 */
const { onCall, HttpsError } = require("firebase-functions/v2/https");
const { defineSecret } = require("firebase-functions/params");
const voice = require("./cchVoiceProfile"); // reused as-is, not modified

const ANTHROPIC_API_KEY = defineSecret("ANTHROPIC_API_KEY");

const REGION = "us-central1";
const MODEL = "claude-sonnet-4-6";

const STUDIO_TEAM_EMAILS = [
  "cindy@cchdesign.com", "cynthia@cchdesign.com", "cynthiacbh@gmail.com",
  "vanessa@cchdesign.com", "vanessaholliday@cchdesign.com", "vholliday@cchdesign.com",
];

function assertStudioTeam(request) {
  if (!request || !request.auth) {
    throw new HttpsError("unauthenticated", "Sign in to use Pepper.");
  }
  const email = String(request.auth.token.email || "").toLowerCase().trim();
  if (!STUDIO_TEAM_EMAILS.includes(email)) {
    throw new HttpsError("permission-denied", "Pepper is for CCH team members only.");
  }
  return email;
}

const PRESET_FRAMES = {
  draft_email: (input) => `Draft a client email. What it needs to say: ${input || "(no detail given, ask one clarifying question)"}`,
  summarize_status: () => "Summarize where this project currently stands based on the activity log below. Plain, short, organized by what's done, what's open, what's waiting on the client.",
  draft_followup: (input) => `Draft a polite follow-up nudge about this: ${input || "the oldest open decision in the activity log"}`,
  whats_next: () => "Based on the activity log below, what needs attention today on this project? Be specific, not generic.",
};

function buildSystemPrompt() {
  // Voice rules restated inline (same content as cchVoiceProfile.js's prompt builders)
  // so cchVoiceProfile.js stays untouched.
  return [
    "You are Pepper, CCH Studio's staff-only assistant, for Cynthia Holloway and Vanessa Holliday.",
    "You are NOT client-facing. You never talk to clients directly and nothing you say is sent to a client without a human reviewing it first.",
    "Because this is staff-only, you may freely reference rates, margins, PO costs, and internal notes when asked.",
    "",
    "Voice, when drafting anything client-facing (emails, tear sheet copy, portal updates):",
    "- Warm, plain-spoken, unpretentious, story-first. \"Designing Your Story.\" Say what actually happened, simply.",
    "- Short sentences. Bullets over paragraphs for anything operational.",
    "- No em dashes, ever. Use commas, periods, colons, or parentheses.",
    "- Kill agency-speak: never \"a period of meaningful refinement,\" \"brought into full resolution,\" \"construction-ready documentation,\" \"architectural permanence,\" \"precision and intentionality,\" or similar.",
    "- Warm, human close when it fits (\"Thanks as always,\" \"Love where this is headed.\").",
    "",
    "When answering status/\"what's next\" questions: be specific and grounded in the activity data you're given below, not generic. If the activity data is thin or missing, say so plainly rather than guessing.",
    "If you don't have enough information to do the task well, ask ONE clarifying question instead of guessing.",
    "Keep replies short by default. This is a working tool, not a document generator.",
  ].join("\n");
}

exports.cchPepper = onCall(
  { region: REGION, invoker: "public", secrets: [ANTHROPIC_API_KEY], timeoutSeconds: 60 },
  async (request) => {
    assertStudioTeam(request);
    const data = request.data || {};
    const action = String(data.action || "").trim();
    const input = String(data.input || "").trim();
    const projectName = String(data.projectName || "").trim();
    // activityDigest: plain-text summary the front-end already built client-side from
    // cch-client-activity.js's existing caFetchProjectActivity(db, projectId). Not
    // re-queried here — reuse the existing engine, don't duplicate it.
    const activityDigest = String(data.activityDigest || "").trim();

    if (!action && !input) {
      throw new HttpsError("invalid-argument", "Ask Pepper a question or pick a preset action.");
    }

    const frame = PRESET_FRAMES[action];
    const task = frame ? frame(input) : input;

    const userMsg = [
      projectName ? "Project: " + projectName : "Project: (none selected)",
      "",
      "Task:",
      task,
      "",
      activityDigest ? "Recent activity on this project:\n" + activityDigest : "(no activity data provided for this request)",
    ].join("\n");

    let resp;
    try {
      resp = await fetch("https://api.anthropic.com/v1/messages", {
        method: "POST",
        headers: {
          "content-type": "application/json",
          "x-api-key": ANTHROPIC_API_KEY.value(),
          "anthropic-version": "2023-06-01",
        },
        body: JSON.stringify({
          model: MODEL,
          max_tokens: 1024,
          system: buildSystemPrompt(),
          messages: [{ role: "user", content: userMsg }],
        }),
      });
    } catch (e) {
      throw new HttpsError("unavailable", "Could not reach Pepper.");
    }

    if (!resp.ok) {
      const t = await resp.text().catch(() => "");
      console.error("[cchPepper] Anthropic error", resp.status, t.slice(0, 500));
      throw new HttpsError("internal", "Pepper's service returned an error " + resp.status);
    }

    const payload = await resp.json();
    const text = (((payload || {}).content || [])[0] || {}).text || "";
    const reply = voice.sanitizeCchVoice(text.trim());

    if (!reply) {
      throw new HttpsError("internal", "Pepper didn't return a reply.");
    }

    return { ok: true, reply };
  }
);
```

### 2. New file `platform/cch-pepper.js`
```js
/**
 * Pepper — floating staff assistant panel.
 * Loaded the same way cch-client-activity.js is (a separate <script src="...">).
 *
 * Hard rule: must NEVER render on a client-facing route. Self-disables on any
 * /#/clientview/ URL regardless of auth state, on top of the Cloud Function's
 * own staff-only check. Two independent guards on purpose.
 */
(function () {
  'use strict';
  if (window._cchPepperLoaded) return;
  window._cchPepperLoaded = true;

  var STUDIO_TEAM_EMAILS = [
    'cindy@cchdesign.com', 'cynthia@cchdesign.com', 'cynthiacbh@gmail.com',
    'vanessa@cchdesign.com', 'vanessaholliday@cchdesign.com', 'vholliday@cchdesign.com',
  ];

  function isClientRoute() { return /#\/clientview\//i.test(window.location.hash || ''); }
  function currentEmail() {
    try { return ((firebase.auth().currentUser || {}).email || '').toLowerCase().trim(); }
    catch (e) { return ''; }
  }
  function isStaff() { return STUDIO_TEAM_EMAILS.indexOf(currentEmail()) !== -1; }
  function projIdFromHash() {
    var m = /#\/project\/([^\/]+)/.exec(window.location.hash || '');
    return m ? m[1] : '';
  }
  function esc(s) {
    var d = document.createElement('div');
    d.textContent = s == null ? '' : String(s);
    return d.innerHTML;
  }

  // Reuses cch-client-activity.js's existing caFetchProjectActivity() directly
  // instead of re-querying Firestore here.
  async function buildActivityDigest(projectId) {
    if (!projectId || typeof window.caFetchProjectActivity !== 'function') return '';
    try {
      var db = (typeof window.db !== 'undefined' && window.db) || firebase.firestore();
      var items = await window.caFetchProjectActivity(db, projectId);
      return items.slice(0, 40).map(function (it) {
        var d = it.data || {};
        var when = d.timestamp && d.timestamp.toDate ? d.timestamp.toDate().toISOString().slice(0, 10) : '';
        return '- [' + when + '] ' + (d.action || 'activity') + ': ' + (d.summary || d.title || d.docType || '');
      }).join('\n');
    } catch (e) {
      console.warn('[cchPepper] activity digest failed', e);
      return '';
    }
  }

  var PRESETS = [
    { key: 'summarize_status', label: 'Summarize status' },
    { key: 'whats_next', label: "What needs attention today" },
    { key: 'draft_email', label: 'Draft client email' },
    { key: 'draft_followup', label: 'Draft a follow-up' },
  ];

  var css = ''
    + '#cch-pepper-btn{position:fixed;right:24px;bottom:24px;z-index:9998;'
    + 'background:#0A1F3D;color:#C8A97E;border:1px solid #C8A97E;border-radius:0;'
    + 'padding:12px 18px;font-family:"DM Sans",system-ui,sans-serif;font-size:13px;'
    + 'letter-spacing:.08em;text-transform:uppercase;cursor:pointer;box-shadow:0 2px 10px rgba(0,0,0,.25);}'
    + '#cch-pepper-btn:hover{background:#0f2a52;}'
    + '#cch-pepper-panel{position:fixed;right:24px;bottom:76px;z-index:9998;width:360px;max-height:70vh;'
    + 'background:#fff;border:1px solid #0A1F3D;border-radius:0;box-shadow:0 4px 24px rgba(0,0,0,.3);'
    + 'display:none;flex-direction:column;font-family:"DM Sans",system-ui,sans-serif;}'
    + '#cch-pepper-panel.open{display:flex;}'
    + '#cch-pepper-head{background:#0A1F3D;color:#fff;padding:14px 16px;font-family:"Playfair Display",serif;'
    + 'font-size:15px;display:flex;justify-content:space-between;align-items:center;}'
    + '#cch-pepper-head small{display:block;color:#C8A97E;font-family:"DM Sans",sans-serif;font-size:11px;margin-top:2px;}'
    + '#cch-pepper-close{cursor:pointer;color:#C8A97E;font-size:16px;line-height:1;background:none;border:none;}'
    + '#cch-pepper-presets{display:flex;flex-wrap:wrap;gap:6px;padding:12px;border-bottom:1px solid #E2E2E2;}'
    + '.cch-pepper-preset{border:1px solid #0A1F3D;color:#0A1F3D;background:#fff;font-size:11px;padding:6px 8px;'
    + 'cursor:pointer;border-radius:0;text-transform:uppercase;letter-spacing:.04em;}'
    + '.cch-pepper-preset:hover{background:#0A1F3D;color:#fff;}'
    + '#cch-pepper-log{flex:1;overflow-y:auto;padding:12px;font-size:13px;color:#0A1F3D;white-space:pre-wrap;}'
    + '.cch-pepper-msg{margin-bottom:10px;}'
    + '.cch-pepper-msg b{color:#C8A97E;font-size:10px;text-transform:uppercase;display:block;margin-bottom:2px;}'
    + '#cch-pepper-inputrow{display:flex;border-top:1px solid #E2E2E2;}'
    + '#cch-pepper-input{flex:1;border:none;padding:10px;font-family:"DM Sans",sans-serif;font-size:13px;}'
    + '#cch-pepper-send{border:none;background:#0A1F3D;color:#C8A97E;padding:0 16px;cursor:pointer;'
    + 'text-transform:uppercase;font-size:11px;letter-spacing:.06em;}';

  function injectStyle() {
    var s = document.createElement('style');
    s.textContent = css;
    document.head.appendChild(s);
  }
  function appendMsg(log, who, text) {
    var row = document.createElement('div');
    row.className = 'cch-pepper-msg';
    row.innerHTML = '<b>' + esc(who) + '</b>' + esc(text);
    log.appendChild(row);
    log.scrollTop = log.scrollHeight;
  }
  async function send(payload, log) {
    appendMsg(log, 'You', payload.input || payload.action.replace(/_/g, ' '));
    var thinking = document.createElement('div');
    thinking.className = 'cch-pepper-msg';
    thinking.innerHTML = '<b>Pepper</b>…';
    log.appendChild(thinking);
    log.scrollTop = log.scrollHeight;
    try {
      var fn = firebase.functions().httpsCallable('cchPepper');
      var res = await fn(payload);
      thinking.innerHTML = '<b>Pepper</b>' + esc((res.data || {}).reply || '(no reply)');
    } catch (e) {
      thinking.innerHTML = '<b>Pepper</b>' + esc('Something went wrong: ' + (e.message || e));
    }
    log.scrollTop = log.scrollHeight;
  }

  function buildPanel() {
    var projectId = projIdFromHash();
    var projectName = (document.querySelector('[data-project-name]') || {}).textContent || '';
    var panel = document.createElement('div');
    panel.id = 'cch-pepper-panel';
    panel.innerHTML =
      '<div id="cch-pepper-head"><div>Pepper<small>' + esc(projectName || 'No project selected') + '</small></div>'
      + '<button id="cch-pepper-close">&times;</button></div>'
      + '<div id="cch-pepper-presets"></div>'
      + '<div id="cch-pepper-log"></div>'
      + '<div id="cch-pepper-inputrow"><input id="cch-pepper-input" placeholder="Ask Pepper anything about this project…" />'
      + '<button id="cch-pepper-send">Send</button></div>';
    document.body.appendChild(panel);

    var presetsWrap = panel.querySelector('#cch-pepper-presets');
    PRESETS.forEach(function (p) {
      var btn = document.createElement('button');
      btn.className = 'cch-pepper-preset';
      btn.textContent = p.label;
      btn.onclick = async function () {
        var digest = await buildActivityDigest(projectId);
        send({ action: p.key, input: '', projectName: projectName, activityDigest: digest }, panel.querySelector('#cch-pepper-log'));
      };
      presetsWrap.appendChild(btn);
    });

    panel.querySelector('#cch-pepper-close').onclick = function () { panel.classList.remove('open'); };

    var input = panel.querySelector('#cch-pepper-input');
    var sendBtn = panel.querySelector('#cch-pepper-send');
    async function submitFree() {
      var val = input.value.trim();
      if (!val) return;
      input.value = '';
      var digest = await buildActivityDigest(projectId);
      send({ action: '', input: val, projectName: projectName, activityDigest: digest }, panel.querySelector('#cch-pepper-log'));
    }
    sendBtn.onclick = submitFree;
    input.addEventListener('keydown', function (e) { if (e.key === 'Enter') submitFree(); });
    return panel;
  }

  function init() {
    if (isClientRoute()) return; // guard #1: never on a client-facing route
    if (!isStaff()) return;      // guard #2: never for a non-staff signed-in user

    injectStyle();
    var btn = document.createElement('button');
    btn.id = 'cch-pepper-btn';
    btn.textContent = 'Pepper';
    document.body.appendChild(btn);

    var panel = null;
    btn.onclick = function () {
      if (!panel) panel = buildPanel();
      panel.classList.toggle('open');
    };
    window.addEventListener('hashchange', function () {
      if (isClientRoute() && panel) panel.classList.remove('open');
    });
  }

  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', init);
  else init();
})();
```
